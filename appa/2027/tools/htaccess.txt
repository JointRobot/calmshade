# Static web app · hosting rules for Apache or LiteSpeed (Hostinger). Ships as .htaccess next to index.html.
# The app is plain HTML + JavaScript modules: browsers refuse to run modules unless .js is served as JavaScript.
AddType text/javascript .js .mjs
AddType application/json .json
AddType audio/mpeg .mp3
AddCharset utf-8 .js .json .html

# compress text files
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/javascript application/javascript application/json image/svg+xml
</IfModule>

# html, js and json are re-checked on every visit (so an update shows at once); images and music are cached for 30 days
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType text/html "access plus 0 seconds"
  ExpiresByType text/javascript "access plus 0 seconds"
  ExpiresByType application/javascript "access plus 0 seconds"
  ExpiresByType application/json "access plus 0 seconds"
  ExpiresByType image/png "access plus 30 days"
  ExpiresByType audio/mpeg "access plus 30 days"
</IfModule>
<IfModule mod_headers.c>
  <FilesMatch "\.(html|js|json)$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
</IfModule>

# no folder listings
Options -Indexes
DirectoryIndex index.html
